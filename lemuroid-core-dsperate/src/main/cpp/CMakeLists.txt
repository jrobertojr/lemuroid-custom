cmake_minimum_required(VERSION 3.22)
project(lemuroid_dsperate LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(DSPERATE_HOST_AARCH64 OFF)
set(DSPERATE_HOST_ARM32 OFF)
set(DSPERATE_JIT OFF)
set(DSPERATE_NEON OFF)
if(ANDROID_ABI STREQUAL "armeabi-v7a")
    set(DSPERATE_HOST_ARM32 ON)
    set(DSPERATE_JIT ON)
    set(DSPERATE_NEON ON)
    # Android uses softfp calling conventions; never import Linux's hard-float ABI.
    add_compile_options(-mfpu=neon-vfpv4)
elseif(ANDROID_ABI STREQUAL "arm64-v8a")
    set(DSPERATE_HOST_AARCH64 ON)
    set(DSPERATE_JIT ON)
    set(DSPERATE_NEON ON)
endif()
set(DSPERATE_GLES OFF CACHE BOOL "" FORCE)
set(DSPERATE_VULKAN OFF CACHE BOOL "" FORCE)
add_compile_options(-O3 -flto=thin -fvisibility=hidden -ffunction-sections -fdata-sections)
add_compile_definitions(NDEBUG)
add_subdirectory(upstream/DSperate/src/core)
add_library(dsperate_libretro_android SHARED upstream/src/libretro.cpp)
target_include_directories(dsperate_libretro_android PRIVATE upstream/include)
target_compile_definitions(dsperate_libretro_android PRIVATE DSPERATE_LR_VERSION="3.3.0-lemuroid1")
target_link_libraries(dsperate_libretro_android PRIVATE dsperate_core log dl)
target_link_options(dsperate_libretro_android PRIVATE "--driver-mode=g++"
    -flto=thin -fuse-ld=lld
    "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/upstream/src/link.T" "-Wl,--gc-sections")
